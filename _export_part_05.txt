import React from 'react';
import { ACTIVITY_BANDS, ALLERGENS, GOALS } from '../data/seed';
import { ActivityBand, AllergenKey, AppState, GoalType, SafetyFlags } from '../types';
import { bandRangeText, calcNutrition, round0, round2, safetyGateError } from '../utils/nutrition';
import { EquationTable } from './EquationTable';
import { SvgRing } from './SvgRing';
import { AlertTriangle, ArrowRight, Check, ShieldAlert } from 'lucide-react';

interface CalculatorWizardProps {
  state: AppState;
  onUpdate: (updater: (prev: AppState) => AppState) => void;
  onFinish: () => void;
}

const Stepper: React.FC<{
  label: string;
  value: number;
  unit: string;
  step: number;
  min?: number;
  max?: number;
  onChange: (val: number) => void;
}> = ({ label, value, unit, step, min = 0, max = 500, onChange }) => (
  <div className="mt-3">
    <label className="block text-xs font-semibold text-[#96938a] mb-1.5">{label}</label>
    <div className="flex items-center bg-[#18181c] border border-[#232329] rounded-xl overflow-hidden">
      <button
        type="button"
        onClick={() => onChange(Math.max(min, round2(value - step)))}
        className="touch-press w-14 h-12 flex items-center justify-center text-xl font-bold text-[#c8a96e] hover:bg-[#1c1a15] cursor-pointer"
        aria-label={`إنقاص ${label}`}
      >
        −
      </button>
      <div className="flex-1 text-center py-1">
        <input
          type="number"
          inputMode="decimal"
          value={value}
          onChange={(e) => {
            const v = parseFloat(e.target.value);
            onChange(Number.isNaN(v) ? 0 : Math.min(max, v));
          }}
          className="w-full bg-transparent border-0 text-center font-num text-xl font-bold text-[#f3f1ec] focus:outline-none tabular-nums"
        />
        <span className="block text-[10px] text-[#66645e] -mt-1">{unit}</span>
      </div>
      <button
        type="button"
        onClick={() => onChange(Math.min(max, round2(value + step)))}
        className="touch-press w-14 h-12 flex items-center justify-center text-xl font-bold text-[#c8a96e] hover:bg-[#1c1a15] cursor-pointer"
        aria-label={`زيادة ${label}`}
      >
        +
      </button>
    </div>
  </div>
);

export const CalculatorWizard: React.FC<CalculatorWizardProps> = ({
  state,
  onUpdate,
  onFinish,
}) => {
  const step = state.ob;
  const gateErr = safetyGateError(state);
  const res = calcNutrition(state);

  const setStep = (n: number) => onUpdate((prev) => ({ ...prev, ob: n }));

  const toggleFlag = (key: keyof SafetyFlags) =>
    onUpdate((prev) => ({
      ...prev,
      fl: { ...prev.fl, [key]: !prev.fl[key] },
    }));

  const toggleAllergen = (key: AllergenKey) =>
    onUpdate((prev) => {
      const exists = prev.al.includes(key);
      const nextAl = exists ? prev.al.filter((a) => a !== key) : [...prev.al, key];
      return { ...prev, al: nextAl };
    });

  const bmi = state.h > 0 ? state.w / Math.pow(state.h / 100, 2) : 0;

  if (step === 0) {
    return (
      <div className="max-w-xl mx-auto px-4 py-8 sm:py-12">
        <div className="text-center space-y-3 mb-8">
          <div className="text-xs font-semibold text-[#c8a96e] tracking-wide">
            الجزائر · معاينة تفاعلية
          </div>
          <h1 className="text-3xl sm:text-4xl font-bold text-[#f3f1ec] text-balance">
            NutriFit DZ
          </h1>
          <p className="text-base sm:text-lg text-[#e6d1a3] font-semibold max-w-md mx-auto text-balance">
            يحسب احتياجك بالغرام، يطبخ وجباتك، ويوصلها لبابك صباحًا.
          </p>
        </div>

        <div className="mb-5 rounded-xl border border-[#c8a96e]/25 bg-[#18181c] px-4 py-3 text-[11px] leading-6 text-[#96938a]">
          <strong className="text-[#e6d1a3]">نسخة معاينة:</strong>{' '}
          هذه الواجهة تختبر رحلة العميل والحسابات محلياً. لم تُربط بعد بخدمة طلبات أو مطبخ أو توصيل مباشر.
        </div>

        <div className="bg-[#111114] border border-[#232329] rounded-2xl p-5 space-y-4 mb-6">
          <div className="text-sm font-bold text-[#f3f1ec]">الرحلة الكاملة في نظام واحد</div>
          <div className="space-y-3 text-sm text-[#96938a] leading-relaxed">
            <div className="flex gap-3">
              <span className="font-num font-bold text-[#c8a96e] tabular-nums">01</span>
              <span>حاسبة علمية شفافة بالمعادلات المنشورة مع بوابة سلامة طبية.</span>
            </div>
            <div className="flex gap-3">
              <span className="font-num font-bold text-[#c8a96e] tabular-nums">02</span>
              <span>تختار وجباتك ومكوناتك بالغرام، وتوزعها على يومك كما تشاء.</span>
            </div>
            <div className="flex gap-3">
              <span className="font-num font-bold text-[#c8a96e] tabular-nums">03</span>
              <span>تُحدّد نافذة التحضير والتوصيل في ملف التشغيل بعد اعتماد المنطقة والطاقة اليومية.</span>
            </div>
            <div className="flex gap-3">
              <span className="font-num font-bold text-[#c8a96e] tabular-nums">04</span>
              <span>هيكل تكلفة قابل للتفصيل: مكونات + مناولة + خدمة + توصيل، وتُعتمد الأسعار الفعلية من الإدارة.</span>
            </div>
          </div>
        </div>

        <div className="space-y-2.5">
          <button
            type="button"
            onClick={() => setStep(1)}
            className="touch-press w-full py-3.5 px-6 rounded-xl font-bold text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] hover:brightness-105 cursor-pointer"
          >
            ابدأ الحاسبة العلمية
          </button>
        </div>

        <p className="text-center text-[11px] text-[#66645e] mt-5">
          أداة تغذية رياضية محسوبة، وليست بديلًا عن الاستشارة الطبية المتخصصة.
        </p>
      </div>
    );
  }

  return (
    <div className="max-w-xl mx-auto px-4 py-6 pb-16">
      <div className="flex items-center gap-1.5 mb-6" aria-label="تقدم الخطوات">
        {[1, 2, 3, 4, 5].map((i) => (
          <div
            key={i}
            className={`flex-1 h-1 rounded-full transition-colors ${
              i <= step ? 'bg-[#c8a96e]' : 'bg-[#232329]'
            }`}
          />
        ))}
      </div>

      {step === 1 && (
        <div className="space-y-4">
          <div>
            <h2 className="text-xl font-bold text-[#f3f1ec]">أولًا: بوابة السلامة</h2>
            <p className="text-xs text-[#96938a] mt-1">
              سلامتك أولوية قصوى قبل أي حسابات. لا نصدر خططًا آلية في الحالات الخاصة.
            </p>
          </div>

          <div className="rounded-xl border border-[#c8a96e]/25 bg-[#18181c] px-3 py-2.5 text-[11px] leading-5 text-[#96938a]">
            <strong className="text-[#e6d1a3]">تنبيه الخصوصية:</strong>{' '}
            بيانات القياسات وإجابات السلامة تُحفظ محلياً في هذا المتصفح فقط. لا توجد مزامنة أو حساب محمي في نسخة المعاينة؛ تجنب إدخال بيانات حساسة على جهاز مشترك.
          </div>

          <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5">
            <label className="block text-xs font-semibold text-[#96938a] mb-1.5">
              الاسم أو اللقب (يُطبع على العلبة)
            </label>
            <input
              type="text"
              value={state.name}
              onChange={(e) => onUpdate((prev) => ({ ...prev, name: e.target.value }))}
              className="w-full bg-[#18181c] border border-[#232329] rounded-xl px-3 py-2.5 text-sm text-[#f3f1ec] focus:border-[#c8a96e] outline-none"
              placeholder="مثال: أمين بن علي"
            />
            <Stepper
              label="العمر"
              value={state.age}
              unit="سنة"
              step={1}
              min={18}
              max={100}
              onChange={(age) => onUpdate((prev) => ({ ...prev, age }))}
            />
          </div>

          <div className="space-y-2">
            <div className="text-xs font-semibold text-[#96938a]">هل ينطبق عليك أي مما يلي؟</div>
            {(
              [
                ['preg', 'حمل أو رضاعة طبيعية'],
                ['kid', 'مرض كلوي معروف'],
                ['dia', 'داء السكري أو أدوية خافضة للسكر'],
                ['ed', 'تاريخ مع اضطرابات الأكل'],
              ] as [keyof SafetyFlags, string][]
            ).map(([key, label]) => {
              const active = state.fl[key];
              return (
                <button
                  key={key}
                  type="button"
                  onClick={() => toggleFlag(key)}
                  className={`touch-press w-full flex items-center justify-between px-4 py-3 rounded-xl border text-right text-sm font-semibold cursor-pointer ${
                    active
                      ? 'bg-[#1a1111] border-[#b86157] text-[#f3f1ec]'
                      : 'bg-[#111114] border-[#232329] text-[#f3f1ec] hover:border-[#c8a96e]/40'
                  }`}
                >
                  <span>{label}</span>
                  <span
                    className={`w-5 h-5 rounded-md flex items-center justify-center border ${
                      active ? 'bg-[#b86157] border-[#b86157] text-[#141108]' : 'border-[#66645e]'
                    }`}
                  >
                    {active && <Check className="w-3.5 h-3.5" />}
                  </span>
                </button>
              );
            })}
          </div>

          {gateErr && (
            <div className="bg-[#1a1111] border border-[#b86157]/60 rounded-2xl p-4 flex gap-3 items-start text-xs text-[#f3f1ec]">
              <ShieldAlert className="w-5 h-5 text-[#b86157] shrink-0 mt-0.5" />
              <div>
                <div className="font-bold text-[#b86157] mb-1">حجب آلي للسلامة</div>
                {gateErr}
              </div>
            </div>
          )}
          {!state.name.trim() && (
            <p className="text-[11px] text-[#c89a56]">أدخل الاسم أو اللقب الذي سيظهر على ملفك قبل المتابعة.</p>
          )}

          <div className="flex gap-2 pt-2">
            <button
              type="button"
              onClick={() => setStep(0)}
              className="touch-press px-4 py-3 rounded-xl bg-[#18181c] border border-[#232329] text-sm font-bold text-[#f3f1ec] flex items-center gap-1 cursor-pointer"
            >
              <ArrowRight className="w-4 h-4" />
              رجوع
            </button>
            <button
              type="button"
              disabled={Boolean(gateErr) || !state.name.trim()}
              onClick={() => setStep(2)}
              className="touch-press flex-1 py-3 rounded-xl font-bold text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] disabled:opacity-40 cursor-pointer"
            >
              التالي: قياسات الجسم
            </button>
          </div>
        </div>
      )}

      {step === 2 && (
        <div className="space-y-4">
          <h2 className="text-xl font-bold text-[#f3f1ec]">قياسات الجسم</h2>
          <div className="rounded-xl border border-[#c8a96e]/25 bg-[#18181c] px-3 py-2.5 text-[11px] leading-5 text-[#96938a]">
            القيم الظاهرة مبدئياً أمثلة للمعاينة فقط. أدخل قياساتك الحقيقية قبل اعتماد أي نتيجة.
          </div>

          <div className="grid grid-cols-2 gap-2">
            {(
              [
                ['m', 'ذكر'],
                ['f', 'أنثى'],
              ] as const
            ).map(([val, lbl]) => (
              <button
                key={val}
                type="button"
                onClick={() => onUpdate((prev) => ({ ...prev, sex: val }))}
                className={`touch-press py-3 rounded-xl border text-sm font-bold cursor-pointer ${
                  state.sex === val
                    ? 'bg-[#1c1a15] border-[#c8a96e] text-[#e6d1a3]'
                    : 'bg-[#111114] border-[#232329] text-[#96938a]'
                }`}
              >
                {lbl}
              </button>
            ))}
          </div>

          <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5">
            <Stepper
              label="الطول"
              value={state.h}
              unit="سم"
              step={1}
              min={120}
              max={230}
              onChange={(h) => onUpdate((prev) => ({ ...prev, h }))}
            />
            <Stepper
              label="الوزن الحالي"
              value={state.w}
              unit="كغ"
              step={0.5}
              min={30}
              max={300}
              onChange={(w) => onUpdate((prev) => ({ ...prev, w }))}
            />
            <Stepper
              label="الوزن المستهدف (للجورنال)"
              value={state.targetWeight}
              unit="كغ"
              step={0.5}
              min={35}
              max={250}
              onChange={(targetWeight) => onUpdate((prev) => ({ ...prev, targetWeight }))}
            />
          </div>

          <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5">
            <div className="text-sm font-bold text-[#c8a96e]">
              الكتلة الخالية من الدهون LBM (اختياري)
            </div>
            <p className="text-xs text-[#96938a] mt-1">
              أدخله فقط إذا كان لديك قياس موثوق مثل DEXA أو InBody بنسبة موثوقية 70% فأكثر لتفعيل معادلة Katch-McArdle.
            </p>
            <Stepper
              label="LBM"
              value={state.lbm}
              unit="كغ (اتركه 0 إن لم يتوفر)"
              step={0.5}
              min={0}
              max={150}
              onChange={(lbm) => onUpdate((prev) => ({ ...prev, lbm }))}
            />
            {state.lbm > 0 && (
              <>
                <div className="mt-2">
                  <label className="text-xs text-[#96938a]">طريقة قياس LBM</label>
                  <select
                    value={state.lbmMethod ?? 'none'}
                    onChange={(e) => onUpdate((prev) => ({
                      ...prev,
                      lbmMethod: e.target.value as 'none' | 'dexa' | 'inbody' | 'other',
                    }))}
                    className="mt-1 w-full rounded-xl border border-[#232329] bg-[#18181c] px-3 py-2.5 text-xs text-[#f3f1ec]"
                  >
                    <option value="none">لم أحدد / غير موثق</option>
                    <option value="dexa">DEXA</option>
                    <option value="inbody">InBody</option>
                    <option value="other">قياس آخر موثق</option>
                  </select>
                </div>
                <Stepper
                  label="موثوقية القياس"
                  value={state.rel}
                  unit="%"
                  step={5}
                  min={0}
                  max={100}
                  onChange={(rel) => onUpdate((prev) => ({ ...prev, rel }))}
                />
                {(!state.lbmMethod || state.lbmMethod === 'none' || state.rel < 70) && (
                  <p className="text-[10px] leading-5 text-[#c89a56] mt-1">
                    ستُستخدم معادلة Mifflin حتى توثّق طريقة قياس LBM وتكون موثوقيتها 70% فأكثر.
                  </p>
                )}
              </>
            )}
          </div>

          <div className="flex gap-2">
            <button
              type="button"
              onClick={() => setStep(1)}
              className="touch-press px-4 py-3 rounded-xl bg-[#18181c] border border-[#232329] text-sm font-bold text-[#f3f1ec] cursor-pointer"
            >
              رجوع
            </button>
            <button
              type="button"
              onClick={() => setStep(3)}
              className="touch-press flex-1 py-3 rounded-xl font-bold text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] cursor-pointer"
            >
              التالي: مستوى النشاط
            </button>
          </div>
        </div>
      )}

      {step === 3 && (
        <div className="space-y-4">
          <div>
            <h2 className="text-xl font-bold text-[#f3f1ec]">مستوى النشاط (4 نطاقات)</h2>
            <p className="text-xs text-[#96938a] mt-1">
              اختر النطاق الأقرب لأسبوعك ثم اضبط المعامل بدقة وفق مرجع FAO/WHO/UNU 2004.
            </p>
          </div>

          <div className="space-y-2">
            {(Object.keys(ACTIVITY_BANDS) as ActivityBand[]).map((key) => {
              const band = ACTIVITY_BANDS[key];
              const active = state.band === key;
              return (
                <button
                  key={key}
                  type="button"
                  onClick={() =>
                    onUpdate((prev) => ({ ...prev, band: key, pal: band.def }))
                  }
                  className={`touch-press w-full text-right p-3.5 rounded-xl border cursor-pointer ${
                    active
                      ? 'bg-[#1c1a15] border-[#c8a96e]'
                      : 'bg-[#111114] border-[#232329] hover:border-[#c8a96e]/40'
                  }`}
                >
                  <div className="flex items-center justify-between">
                    <span className="font-bold text-sm text-[#f3f1ec]">{band.label}</span>
                    <span className="font-num text-xs font-bold text-[#c8a96e] tabular-nums">
                      {bandRangeText(key)}
                    </span>
                  </div>
                  <div className="text-xs text-[#96938a] mt-1">{band.desc}</div>
                </button>
              );
            })}
          </div>

          <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5">
            <div className="flex items-center justify-between text-xs mb-2">
              <span className="text-[#96938a] font-semibold">ضبط دقيق للمعامل</span>
              <span className="font-num text-base font-bold text-[#e6d1a3] tabular-nums">
                {state.pal.toFixed(2)}
              </span>
            </div>
            <input
              type="range"
              min={ACTIVITY_BANDS[state.band].min}
              max={ACTIVITY_BANDS[state.band].max}
              step={0.01}
              value={state.pal}
              onChange={(e) =>
                onUpdate((prev) => ({ ...prev, pal: parseFloat(e.target.value) }))
              }
              className="w-full accent-[#c8a96e] cursor-pointer"
            />
            <div className="flex justify-between font-num text-[10px] text-[#66645e] mt-1 tabular-nums">
              <span>{ACTIVITY_BANDS[state.band].min.toFixed(2)}</span>
              <span>{ACTIVITY_BANDS[state.band].max.toFixed(2)}</span>
            </div>
          </div>

          <div className="flex gap-2">
            <button
              type="button"
              onClick={() => setStep(2)}
              className="touch-press px-4 py-3 rounded-xl bg-[#18181c] border border-[#232329] text-sm font-bold text-[#f3f1ec] cursor-pointer"
            >
              رجوع
            </button>
            <button
              type="button"
              onClick={() => setStep(4)}
              className="touch-press flex-1 py-3 rounded-xl font-bold text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] cursor-pointer"
            >
              التالي: الهدف والحساسيات
            </button>
          </div>
        </div>
      )}

      {step === 4 && (
        <div className="space-y-4">
          <h2 className="text-xl font-bold text-[#f3f1ec]">الهدف والحساسيات</h2>

          <div className="space-y-2">
            {(Object.keys(GOALS) as GoalType[]).map((key) => {
              const g = GOALS[key];
              const blockedByBmi = key === 'cut' && bmi < 18.5;
              const active = state.goal === key;
              return (
                <button
                  key={key}
                  type="button"
                  disabled={blockedByBmi}
                  onClick={() =>
                    onUpdate((prev) => ({ ...prev, goal: key, pct: g.pct }))
                  }
                  className={`touch-press w-full text-right p-3.5 rounded-xl border cursor-pointer disabled:opacity-40 disabled:cursor-not-allowed ${
                    active
                      ? 'bg-[#1c1a15] border-[#c8a96e]'
                      : 'bg-[#111114] border-[#232329] hover:border-[#c8a96e]/40'
                  }`}
                >
                  <div className="flex items-center justify-between">
                    <span className="font-bold text-sm text-[#f3f1ec]">{g.label}</span>
                    <span className="font-num text-xs font-bold text-[#c8a96e] tabular-nums">
                      {g.multLabel} · {g.proteinPerKg}غ/كغ
                    </span>
                  </div>
                  <div className="text-xs text-[#96938a] mt-1">
                    {blockedByBmi ? 'محجوب لأن BMI أقل من 18.5' : g.desc}
                  </div>
                </button>
              );
            })}
          </div>

          <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5">
            <Stepper
              label="تعديل نسبة الفائض أو العجز (حد الأمان ±25%)"
              value={state.pct}
              unit="%"
              step={1}
              min={-25}
              max={25}
              onChange={(pct) => onUpdate((prev) => ({ ...prev, pct }))}
            />
          </div>

          <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5 space-y-3">
            <div className="text-xs font-bold text-[#c8a96e]">الحساسيات الغذائية</div>
            <div className="flex flex-wrap gap-2">
              {(Object.keys(ALLERGENS) as AllergenKey[]).map((k) => {
                const on = state.al.includes(k);
                return (
                  <button
                    key={k}
                    type="button"
                    onClick={() => toggleAllergen(k)}
                    className={`touch-press px-3.5 py-2 rounded-xl border text-xs font-bold cursor-pointer ${
                      on
                        ? 'bg-[#1c1a15] border-[#c8a96e] text-[#e6d1a3]'
                        : 'bg-[#18181c] border-[#232329] text-[#96938a]'
                    }`}
                  >
                    {on ? `✓ ${ALLERGENS[k]}` : ALLERGENS[k]}
                  </button>
                );
              })}
            </div>

            <div className="pt-2 border-t border-[#232329]">
              <div className="text-xs font-bold text-[#c8a96e] mb-2">
                مكمل البروتين (اختياري)
              </div>
              <div className="grid grid-cols-2 gap-2">
                <button
                  type="button"

                  label="كمية البروتين الصافي من المكمل يوميًا"

                  unit="غ / يوم (يُخصم من حاجة الطعام)"
                  step={5}
                  min={0}
                  max={150}
                  onChange={(supg) => onUpdate((prev) => ({ ...prev, supg }))}

              )}
            </div>

            >
              احسب احتياجي العلمي
            </button>
          </div>
        </div>
      )}

      {step === 5 && (
        <div className="space-y-4">
          {res.err ? (
            <div className="bg-[#1a1111] border border-[#b86157] rounded-2xl p-4 flex gap-3 text-sm text-[#f3f1ec]">

            </div>
          ) : (

                <SvgRing progress={1} size={184} strokeWidth={8} color="#c8a96e" trackColor="#232329">
                  <div className="text-4xl font-bold text-[#e6d1a3] tabular-nums">
                    {round0(res.kcal)}
                  </div>
                  <div className="text-xs text-[#96938a]">kcal / يوم</div>
                </SvgRing>
              </div>

              <div className="grid grid-cols-3 gap-2">
                <div className="bg-[#111114] border border-[#232329] rounded-2xl p-3 text-center">
                  <div className="text-xl font-bold text-[#f3f1ec] tabular-nums">
                    {round0(res.P)} غ
                  </div>
                  <div className="text-[11px] text-[#96938a]">بروتين</div>
                </div>
                <div className="bg-[#111114] border border-[#232329] rounded-2xl p-3 text-center">
                  <div className="text-xl font-bold text-[#f3f1ec] tabular-nums">
                    {round0(res.C)} غ
                  </div>
                  <div className="text-[11px] text-[#96938a]">كارب</div>
                </div>
                <div className="bg-[#111114] border border-[#232329] rounded-2xl p-3 text-center">
                  <div className="text-xl font-bold text-[#f3f1ec] tabular-nums">
                    {round0(res.F)} غ
                  </div>
                  <div className="text-[11px] text-[#96938a]">دهون</div>
                </div>
              </div>

              <EquationTable state={state} result={res} />
            </>
          )}

          <div className="flex gap-2">
            <button
              type="button"
              onClick={() => setStep(4)}
              className="touch-press px-4 py-3 rounded-xl bg-[#18181c] border border-[#232329] text-sm font-bold text-[#f3f1ec] cursor-pointer"
            >
              تعديل
            </button>
            <button
              type="button"
              disabled={Boolean(res.err)}
              onClick={onFinish}
              className="touch-press flex-1 py-3.5 rounded-xl font-bold text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] disabled:opacity-40 cursor-pointer"
            >
              ادخل إلى التطبيق
            </button>
          </div>
        </div>
      )}
    </div>
  );
};

const STAGE_LABELS: Record<NonNullable<Order['productionStage']>, string> = {

export const CartView: React.FC<CartViewProps> = ({

  onUpdate,
  onNavigate,
  onSelectMealSlot,
  onToast,
  apiEnabled = false,
  authToken,
}) => {
  const [confirmModal, setConfirmModal] = useState<Order | null>(null);
  const [submitting, setSubmitting] = useState(false);
  const res = calcNutrition(state);
  const totals = calcCartTotals(state);
  const pricingCfg = state.pricingConfig || DEFAULT_PRICING;
  const dlv = getNextDelivery(pricingCfg.cutoffHour);
  const activeZones = state.zones.filter((z) => z.active);
  const selectedZone = activeZones.find((z) => z.id === state.zoneId) ?? activeZones[0] ?? getSelectedZone(state);
  const zoneOptions = activeZones;
  const pricing = calcCartPrice(totals.price, state.n, selectedZone.fee, pricingCfg);

  if (res.err) {
    return (
      <div className="bg-[#1a1111] border border-[#b86157]/50 rounded-2xl p-4 text-sm text-[#f3f1ec]">
        {res.err}
      </div>
    );
  }

  const removeItem = (uid: string) => {
    const backup = [...state.meals];
    onUpdate((prev) => ({
      ...prev,
      meals: prev.meals.filter((m) => m.uid !== uid),
    }));
    onToast('حُذف الصنف من السلة', () => {
      onUpdate((prev) => ({ ...prev, meals: backup }));
    });
  };

  const handleAutoDistribute = () => {
    onUpdate((prev) => ({
      ...prev,
      meals: autoDistributeMeals(prev.meals, prev.n, prev.foods, prev.dishes),
    }));
    onToast('تم توزيع الأصناف تلقائيًا على وجبات اليوم');
  };

  const updateMealName = (slotIdx: number, newName: string) => {
    onUpdate((prev) => {
      const names = [...(prev.mealNames || DEFAULT_MEAL_NAMES)];
      while (names.length < 5) {
        names.push(DEFAULT_MEAL_NAMES[names.length] || `الوجبة ${names.length + 1}`);
      }
      names[slotIdx] = newName;
      return { ...prev, mealNames: names };
    });
  };

  const cancelOrder = async (order: Order) => {
    if (apiEnabled) {
      if (!authToken) { onToast('انتهت جلسة الدخول؛ سجّل الدخول مجدداً.'); return; }
      try {
        await apiRequest('/api/orders/' + order.id,authToken,{method:'DELETE'});
      } catch (error) {
        onToast(error instanceof Error ? error.message : 'تعذر إلغاء الطلب.');
        return;
      }
    }
    onUpdate((prev) => ({
      ...prev,
      ord: prev.ord.map((item) => item.id === order.id ? {...item,status:'X'} : item),
    }));
    onToast('تم إلغاء الطلب.');
  };

  const handlePlaceOrder = async () => {
    if (submitting) return;
    if (!state.name.trim()) {
      onToast('أدخل الاسم أو اللقب في ملفك قبل حفظ الطلب');
      return;

    if (activeZones.length === 0) {
      onToast('لا توجد منطقة خدمة معتمدة بعد؛ لا يمكن حفظ طلب توصيل حتى يحددها المسؤول');

    if (!selectedZone.active) {
      onToast('اختر منطقة خدمة معتمدة قبل المتابعة');

    if (state.meals.length === 0) {
      onToast('السلة فارغة؛ أضف وجباتك أولاً');

    if (emptySlot !== undefined) {

      } else if (m.kind === 'dish') {
        const d = state.dishes.find((x) => x.id === m.id);
        if (!d) {
          onToast('أحد الأطباق لم يعد موجودًا في القائمة؛ احذفه ثم أعد اختياره');

          return;

          onToast('توجد مكونات ناقصة في أحد الأطباق؛ راجع قائمة الطعام');

        }
        if (d.ing.some(([foodId]) => state.foods.find((food) => food.id === foodId)?.available === false)) {
          onToast('أحد مكونات الطبق غير متوفر اليوم');
          return;

        if (getBlockedAllergensForDish(d, state.foods, state.al).length > 0) {
          onToast('تحتوي السلة على طبق يسبب لك حساسية');

          return;
        }
        for (const [foodId] of ingredients) {

            return;

    }

    const usedMeals = state.ord
      .filter((o) => o.date === dlv.date && o.status === 'C')
      .reduce((acc, o) => acc + o.n, 0);
    const status: 'C' | 'W' = usedMeals + state.n > state.cap ? 'W' : 'C';

    const items = state.meals.map((m) => {

      const slotIdx = clamp(m.slot || 0, 0, state.n - 1);
      let label = m.name || 'وجبة مخصصة';
      if (m.kind === 'food') {
        label = state.foods.find((x) => x.id === m.id)?.name || 'مكوّن';
      } else if (m.kind === 'dish') {
        label = state.dishes.find((x) => x.id === m.id)?.name || 'طبق';

        kcal: round0(mac.kcal),
        p: round0(mac.p),
        slot: slotIdx,
        slotName: getMealSlotName(state, slotIdx),
        rawIngredients: mac.rawIngredients,
      };

      date: dlv.date,

            items: state.meals.map((item) => ({

                ? (item.ing ?? []).map(([foodId, grams]) => ({ foodId, grams }))

              slot: clamp(item.slot || 0, 0, state.n - 1),
              slotName: getMealSlotName(state, clamp(item.slot || 0, 0, state.n - 1)),
            })),

        });

          id: response.order.id,
          date: String(response.order.delivery_date).slice(0, 10),

          zoneFee: Number(response.order.delivery_fee_dzd),
          ingCost: Number(response.order.ingredient_cost_dzd),
          handling: Number(response.order.handling_fee_dzd),

          p: Number(response.order.protein_total),
          c: Number(response.order.carbs_total),
          f: Number(response.order.fat_total),
          status: response.order.order_status === 'confirmed' ? 'C' : 'W',
          productionStage: response.order.production_stage ?? undefined,

      } catch (error) {
        onToast(error instanceof Error ? error.message : 'تعذر إرسال الطلب إلى الخادم.');

          <div className="text-sm font-bold text-[#f3f1ec]">سلة وجبات الغد فارغة</div>
          <p className="text-xs text-[#96938a]">
            اختر أطباقًا مختلفة للوجبة الصباحية وبعد التمرين والعشاء من القائمة.
          </p>
          <button
            type="button"
            onClick={() => onNavigate('menu')}
            className="touch-press px-6 py-3 rounded-xl font-bold text-sm text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] cursor-pointer"

            تصفح الوجبات

        </div>
      ) : (
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-4 items-start">
          {/* Left Column: Macro Bars + Daily Meal Sections */}
          <div className="lg:col-span-7 space-y-4">
            {/* Purposeful Executive Macro Bars */}
            <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5 space-y-2.5">
              {[
                { label: 'سعرات', val: totals.kcal, target: res.kcal, barColor: '#c8a96e', unit: '' },
                { label: 'بروتين', val: totals.p, target: res.fp, barColor: '#e6d1a3', unit: 'غ' },
                { label: 'كارب', val: totals.c, target: res.C, barColor: '#6b9c7e', unit: 'غ' },
                { label: 'دهون', val: totals.f, target: res.F, barColor: '#a98c72', unit: 'غ' },
              ].map((b) => {
                const pct = b.target > 0 ? clamp((b.val / b.target) * 100, 0, 100) : 0;
                return (

                    onUpdate((prev) => ({
                      ...prev,
                      n: nextN,
                      activeSlot: Math.min(prev.activeSlot ?? 0, nextN - 1),

                        ...m,
                        slot: Math.min(m.slot, nextN - 1),

                    }));

                  aria-label="تقليل وجبة"

                  {state.n}

                    onUpdate((prev) => ({ ...prev, n: Math.min(5, prev.n + 1) }))
                  }
                  className="touch-press w-9 h-9 rounded-xl bg-[#18181c] border border-[#232329] flex items-center justify-center text-[#f3f1ec] disabled:opacity-35 cursor-pointer"
                  aria-label="إضافة وجبة"

              </div>
            </div>

            {/* Grouped by Distinct Daily Meal */}
            <div className="space-y-3">
              {Array.from({ length: state.n }, (_, slotIdx) => {
                const slotMeals = state.meals.filter(
                  (m) => clamp(m.slot || 0, 0, state.n - 1) === slotIdx
                );
                const slotTotals = slotMeals.reduce(
                  (acc, m) => {
                    const mac = cartItemMacros(m, state.foods, state.dishes);
                    acc.kcal += mac.kcal;
                    acc.p += mac.p;
                    acc.price += mac.price;
                    return acc;
                  },
                  { kcal: 0, p: 0, price: 0 }
                );

                return (
                  <div
                    key={slotIdx}
                    className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5 space-y-2.5"
                  >
                    <div className="flex items-center justify-between gap-2 pb-2 border-b border-[#232329]">
                      <input
                        type="text"
                        value={getMealSlotName(state, slotIdx)}
                        onChange={(e) => updateMealName(slotIdx, e.target.value)}
                        className="bg-transparent text-sm font-bold text-[#e6d1a3] focus:outline-none focus:border-b focus:border-[#c8a96e] min-w-0 flex-1"
                        aria-label={`اسم الوجبة ${slotIdx + 1}`}
                      />
                      <div className="text-xs text-[#96938a] tabular-nums shrink-0">
                        {round0(slotTotals.kcal)} kcal · <span className="text-[#c8a96e]">P{round0(slotTotals.p)}</span>
                      </div>
                    </div>

                    {slotMeals.length === 0 ? (
                      <div className="flex items-center justify-between py-2 text-xs text-[#66645e]">
                        <span>لا توجد أصناف مخصصة لهذه الوجبة بعد.</span>
                        <button
                          type="button"
                          onClick={() => onSelectMealSlot(slotIdx)}
                          className="touch-press text-[#c8a96e] font-semibold hover:underline cursor-pointer"
                        >
                          + أضف طبقًا
                        </button>
                      </div>
                    ) : (
                      <div className="divide-y divide-[#232329]">
                        {slotMeals.map((m) => {
                          const mac = cartItemMacros(m, state.foods, state.dishes);
                          let name = m.name || 'وجبة مخصصة';
                          let isJuice = false;
                          if (m.kind === 'food') {
                            const f = state.foods.find((x) => x.id === m.id);
                            if (f) {
                              name = f.name;
                              isJuice = f.cat === 'juice';
                            }
                          } else if (m.kind === 'dish') {
                            const d = state.dishes.find((x) => x.id === m.id);
                            if (d) {
                              name = d.name;
                            }
                          }
                          const customBreakdown =
                            m.kind === 'custom' && Array.isArray(m.ing) && m.ing.length > 0
                              ? m.ing
                                  .map(([fId, g]) => {
                                    const fd = state.foods.find((x) => x.id === fId);
                                    if (!fd) return null;
                                    return `${fd.name} (${round0(g)}${fd.cat === 'juice' ? 'مل' : 'غ'})`;
                                  })
                                  .filter(Boolean)
                                  .join(' + ')
                              : '';

                          return (
                            <div
                              key={m.uid}
                              className="py-2.5 flex flex-wrap sm:flex-nowrap items-center gap-2"
                            >
                              <div className="flex-1 min-w-[140px]">
                                <div className="text-xs sm:text-sm font-semibold text-[#f3f1ec]">
                                  {name}
                                </div>
                                {customBreakdown && (
                                  <div className="text-[11px] text-[#96938a] mt-0.5 leading-relaxed">
                                    {customBreakdown}
                                  </div>
                                )}
                                <div className="text-[11px] text-[#96938a] tabular-nums mt-0.5">
                                  {round0(mac.kcal)} kcal ·{' '}
                                  <span className="text-[#c8a96e]">P{round0(mac.p)}</span>{' '}
                                  <span className="text-[#6b9c7e]">C{round0(mac.c)}</span>{' '}
                                  <span className="text-[#a98c72]">F{round0(mac.f)}</span> ·{' '}
                                  <span className="text-[#e6d1a3] font-bold">{round1(mac.price)} دج</span>
                                </div>
                              </div>

                              {m.kind === 'food' ? (
                                <div className="flex items-center gap-1">
                                  <input
                                    type="number"
                                    value={m.grams || 0}
                                    onChange={(e) => {
                                      const g = Math.max(0, parseFloat(e.target.value) || 0);
                                      onUpdate((prev) => ({
                                        ...prev,
                                        meals: prev.meals.map((item) =>
                                          item.uid === m.uid ? { ...item, grams: g } : item
                                        ),
                                      }));
                                    }}
                                    className="w-16 bg-[#18181c] border border-[#232329] rounded-lg px-2 py-1 text-center text-xs text-[#f3f1ec] tabular-nums focus:border-[#c8a96e] outline-none"
                                  />
                                  <span className="text-[11px] text-[#66645e]">
                                    {isJuice ? 'مل' : 'غ'}
                                  </span>
                                </div>
                              ) : (
                                <span className="text-xs text-[#96938a] px-2 tabular-nums">
                                  {m.kind === 'dish'
                                    ? m.mult === 0.75
                                      ? 'حصة خفيفة'
                                      : m.mult === 1.5
                                        ? 'حصة كبيرة'
                                        : 'حصة قياسية'
                                    : `${round0(mac.grams)}غ`}
                                </span>
                              )}

                              <select
                                value={slotIdx}
                                onChange={(e) => {
                                  const nextSlot = Number(e.target.value);
                                  onUpdate((prev) => ({
                                    ...prev,
                                    meals: prev.meals.map((item) =>
                                      item.uid === m.uid ? { ...item, slot: nextSlot } : item
                                    ),
                                  }));
                                }}
                                className="bg-[#18181c] border border-[#232329] rounded-lg px-2 py-1.5 text-[11px] text-[#f3f1ec] max-w-[135px]"
                              >
                                {Array.from({ length: state.n }, (_, s) => (
                                  <option key={s} value={s}>
                                    {getMealSlotName(state, s)}
                                  </option>
                                ))}
                              </select>

                              <button
                                type="button"
                                onClick={() => removeItem(m.uid)}
                                className="touch-press p-1.5 text-[#96938a] hover:text-[#b86157] rounded-lg cursor-pointer"
                                aria-label="حذف"
                              >
                                <Trash2 className="w-4 h-4" />
                              </button>
                            </div>
                          );
                        })}
                      </div>
                    )}
                  </div>
                );
              })}
            </div>

            <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5">
              <div className="text-sm font-bold text-[#f3f1ec] mb-3">ملخص توزيع اليوم</div>
              <MealSlotsSummary state={state} onSelectSlot={onSelectMealSlot} />
            </div>
          </div>

          {/* Right Column: Delivery Zone, Note & Transparent Invoice */}
          <div className="lg:col-span-5 space-y-4">
            <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 sm:p-5 space-y-4">
              <div>
                <div className="text-sm font-bold text-[#f3f1ec] mb-2">منطقة التوصيل أو الاستلام</div>
                <div className="space-y-2">
                  {zoneOptions.length === 0 ? (
                    <div className="rounded-xl border border-[#b86157]/40 bg-[#1a1111] p-3 text-xs leading-6 text-[#e6b4aa]">
                      لا توجد حالياً منطقة توصيل أو نقطة استلام معتمدة. يمكنك مواصلة معاينة الوجبات، لكن تأكيد الطلب متوقف حتى اعتماد نطاق الخدمة.

                        <div className="flex items-center justify-between text-xs font-bold">
                          <span className="text-[#f3f1ec]">{z.title}</span>

                  })}
                </div>
              </div>

              <div>
                <label className="block text-xs font-semibold text-[#96938a] mb-1.5">
                  ملاحظة للمطبخ (اختياري)

            onClick={(e) => e.stopPropagation()}
          >
            <div className="flex justify-center">
              {confirmModal.status === 'C' ? (
                <CheckCircle2 className="w-11 h-11 text-[#6b9c7e]" />
              ) : (
                <Clock className="w-11 h-11 text-[#c89a56]" />
              )}
            </div>
            <h3 className="text-lg font-bold text-[#f3f1ec]">
              {apiEnabled
                ? confirmModal.status === 'C' ? 'تم تسجيل طلبك بنجاح' : 'تم تسجيل طلبك في قائمة الانتظار'
                : confirmModal.status === 'C' ? 'تم حفظ طلب المعاينة' : 'تم حفظ طلب المعاينة في قائمة الانتظار'}
            </h3>
            <p className="text-xs text-[#96938a] leading-relaxed">
              {apiEnabled
                ? confirmModal.status === 'C'
                  ? 'تم تسجيل الطلب في قاعدة بيانات الخدمة. سيظهر لفريق التشغيل وفق السعة المعتمدة، وتبقى حالة الدفع نقداً عند الاستلام.'
                  : 'سُجل الطلب في قائمة الانتظار على الخادم. لا يبدأ التحضير إلا بعد تأكيد الإدارة توفر السعة.'
                : confirmModal.status === 'C'
                  ? 'هذا سجل معاينة محفوظ على هذا الجهاز فقط؛ لم يُرسل إلى المطبخ ولم يُنشأ توصيل فعلي.'
                  : 'هذه محاكاة محلية لقائمة الانتظار؛ لا توجد خدمة إرسال أو إشعارات للمطبخ.'}
            </p>
            <div className="bg-[#18181c] border border-[#232329] rounded-xl p-3 text-xs flex justify-between tabular-nums">
              <span className="text-[#96938a]">رقم الطلب: {confirmModal.id}</span>
              <span className="text-[#e6d1a3] font-bold">{confirmModal.total} دج</span>
            </div>
            <button
              type="button"
              onClick={() => {
                setConfirmModal(null);
                onNavigate('home');
              }}
              className="touch-press w-full py-3 rounded-xl font-bold text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] cursor-pointer"
            >
              العودة إلى شاشة اليوم
            </button>
          </div>
        </div>
      )}
    </div>
  );
};

            className="touch-press min-h-[44px] px-5 py-2.5 rounded-xl font-bold text-sm text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] flex items-center gap-2 cursor-pointer shrink-0 whitespace-nowrap"

            initial={{ opacity: 0 }}

            transition={{ duration: 0.18 }}
            className="fixed inset-0 z-50 bg-[#202722]/45 backdrop-blur-sm flex items-end sm:items-center sm:justify-end"
            onClick={(e) => {
              if (e.target === e.currentTarget) setSelectedFoodId(null);
            }}
          >
            <motion.aside
              initial={{ x: -32, opacity: 0 }}
              animate={{ x: 0, opacity: 1 }}
              exit={{ x: -32, opacity: 0 }}
              transition={{ duration: 0.22, ease: [0.16, 1, 0.3, 1] }}
              aria-label="تخصيص الكمية والخلط"
              className="w-full max-w-lg h-[94dvh] sm:h-full sm:max-h-full rounded-t-[28px] sm:rounded-none bg-[#fffefb] border border-[#e3dfd4] sm:border-y-0 sm:border-l-0 flex flex-col shadow-2xl overflow-hidden"
            >
              {(() => {
                const isJuice = selectedFood.cat === 'juice';

                } else if (dom.key === 'F' && selectedFood.per100.f >= 3 && gaps.F > 2) {

                    Math.round(((gaps.F / selectedFood.per100.f) * 100) / gramCfg.step) *

                  );

                }

                const remKcalAfter = Math.max(0, res.kcal - (cartTotals.kcal + mac.kcal));

                        aria-label="إغلاق"
                      >
                        <X className="w-4 h-4" />
                      </button>
                    </div>

                    {/* Drawer Content */}
                    <div className="flex-1 overflow-y-auto p-4 sm:p-5 space-y-5">
                      {/* Frosted Editorial Food Banner */}
                      <div className="relative h-40 rounded-2xl overflow-hidden border border-[#232329] bg-[#18181c]">
                        <img
                          ref={drawerImageRef}
                          src={getFoodImage(selectedFood)}
                          alt={selectedFood.name}
                          referrerPolicy="no-referrer"
                          className="w-full h-full object-cover"
                        />
                        <div className="absolute inset-0 bg-gradient-to-t from-[#09090b] via-[#09090b]/55 to-transparent" />
                        <div className="absolute bottom-3 right-4 left-4 flex items-end justify-between gap-3">
                          <div>
                            <div className="text-[11px] text-[#c8a96e]">
                              {CATEGORY_LABELS[selectedFood.cat]} · {formatProvenanceShort(selectedFood)}
                            </div>
                            <div className="text-xs text-[#96938a] mt-0.5 tabular-nums">
                              لكل 100{unit}: {round0(selectedFood.per100.kcal)} kcal · P{round1(selectedFood.per100.p)} · C{round1(selectedFood.per100.c)} · F{round1(selectedFood.per100.f)}
                            </div>
                          </div>
                          <div className="text-left shrink-0 tabular-nums">
                            <div className="text-sm font-bold text-[#e6d1a3]">
                              {round1(selectedFood.price)} دج
                            </div>
                            <div className="text-[10px] text-[#96938a]">/ 100{unit}</div>
                          </div>
                        </div>
                      </div>

                      {selectedFood.nutritionDataStatus !== 'verified' && (
                        <div className="text-[11px] leading-5 text-[#c89a56] border-b border-[#232329] pb-3">
                          تنبيه علمي: القيم الغذائية لهذا الصنف في مرحلة المعاينة الأولية ولم تُعتمد بعد بسجل مخبري رسمي.
                        </div>
                      )}

                      {/* Live nutrition and price — pinned above quantity controls */}
                      <div className="sticky top-0 z-20 -mx-1 rounded-2xl border border-[#d9e1d8] bg-[#fffefb]/95 p-4 shadow-[0_8px_24px_rgba(32,39,34,0.08)] backdrop-blur-md space-y-2.5">
                        <div className="flex items-center justify-between text-xs">
                          <span className="font-bold text-[#f3f1ec]">الأثر الغذائي والتكلفة</span>
                          <span className="text-sm font-bold text-[#e6d1a3] tabular-nums">
                            {round1(mac.price)} دج ({round2(perUnitPrice)}/{unit})
                          </span>
                        </div>

                        <div className="grid grid-cols-4 gap-2 text-center bg-[#18181c] rounded-xl py-2.5 px-1 tabular-nums text-xs">
                          <div>
                            <div className="font-bold text-[#e6d1a3]">+{round0(mac.kcal)}</div>
                            <div className="text-[10px] text-[#96938a]">سعرة</div>
                          </div>
                          <div>
                            <div className="font-bold text-[#c8a96e]">+{round1(mac.p)}غ</div>
                            <div className="text-[10px] text-[#96938a]">بروتين</div>
                          </div>
                          <div>
                            <div className="font-bold text-[#6b9c7e]">+{round1(mac.c)}غ</div>
                            <div className="text-[10px] text-[#96938a]">كارب</div>
                          </div>
                          <div>
                            <div className="font-bold text-[#a98c72]">+{round1(mac.f)}غ</div>
                            <div className="text-[10px] text-[#96938a]">دهون</div>
                          </div>
                        </div>

                        <div className="text-[11px] text-[#96938a] tabular-nums flex items-center justify-between pt-1">
                          <span>المتبقي لليوم بعد الإضافة:</span>
                          <b className="text-[#f3f1ec]">
                            {round0(remKcalAfter)} kcal · P{round0(remPAfter)} C{round0(remCAfter)} F{round0(remFAfter)}
                          </b>
                        </div>
                      </div>


                      {/* 1. Precision Quantity Selector */}
                      <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 space-y-3.5">
                        <div className="flex items-center justify-between gap-2">
                          <span className="text-xs font-bold text-[#f3f1ec]">
                            1. تحديد الكمية ({unit})
                          </span>
                          {smartFill && (
                            <button
                              type="button"
                              onClick={() => setGrams(smartFill.grams)}
                              className="touch-press px-3 py-1.5 rounded-lg bg-[#1c1a15] border border-[#c8a96e]/45 text-[11px] font-bold text-[#e6d1a3] cursor-pointer tabular-nums"
                            >
                              {smartFill.label}
                            </button>
                          )}
                        </div>

                        <div className="flex items-center bg-[#18181c] border border-[#232329] rounded-xl overflow-hidden">
                          <button
                            type="button"
                            onClick={() => setGrams((g) => Math.max(0, g - gramCfg.step))}
                            className="touch-press w-14 h-12 text-xl font-bold text-[#c8a96e] hover:bg-[#232329] cursor-pointer"
                            aria-label="إنقاص"
                          >
                            −
                          </button>
                          <input
                            type="number"
                            value={grams}
                            onChange={(e) => setGrams(Math.max(0, parseFloat(e.target.value) || 0))}
                            className="flex-1 bg-transparent text-center text-2xl font-bold text-[#f3f1ec] tabular-nums focus:outline-none"
                            aria-label="الكمية"
                          />
                          <button
                            type="button"
                            onClick={() => setGrams((g) => g + gramCfg.step)}
                            className="touch-press w-14 h-12 text-xl font-bold text-[#c8a96e] hover:bg-[#232329] cursor-pointer"
                            aria-label="زيادة"
                          >
                            +
                          </button>
                        </div>

                        <div>
                          <input
                            type="range"
                            min={gramCfg.min}
                            max={gramCfg.max}
                            step={gramCfg.step}
                            value={clamp(grams, gramCfg.min, gramCfg.max)}
                            onChange={(e) => setGrams(parseFloat(e.target.value) || 0)}
                            className="w-full accent-[#c8a96e] cursor-pointer"
                          />
                          <div className="flex justify-between text-[10px] text-[#66645e] tabular-nums mt-1">
                            <span>{gramCfg.min}{unit}</span>
                            <span>{gramCfg.max}{unit}</span>
                          </div>
                        </div>

                        <div className="grid grid-cols-4 gap-2">
                          {gramCfg.presets.map((preset) => (
                            <button
                              key={preset}
                              type="button"
                              onClick={() => setGrams(preset)}
                              className={`touch-press min-h-[40px] py-1.5 rounded-lg border text-xs font-semibold tabular-nums cursor-pointer ${
                                grams === preset
                                  ? 'bg-[#1c1a15] border-[#c8a96e] text-[#e6d1a3]'
                                  : 'bg-[#18181c] border-[#232329] text-[#f3f1ec] hover:border-[#c8a96e]/40'
                              }`}
                            >
                              {preset}{unit}
                            </button>
                          ))}
                        </div>
                      </div>

                      {/* 2. Standalone vs. Mix With Another Food */}
                      <div className="bg-[#111114] border border-[#232329] rounded-2xl p-4 space-y-3">
                        <div className="text-xs font-bold text-[#f3f1ec] flex items-center gap-1.5">
                          <Layers className="w-4 h-4 text-[#c8a96e]" />
                          <span>2. طريقة الإضافة للوجبة</span>
                        </div>

                        <div className="grid grid-cols-2 gap-2">
                          <button
                            type="button"
                            onClick={() => setPortionMode('standalone')}
                            className={`touch-press min-h-[52px] p-3 rounded-xl border text-right cursor-pointer ${
                              portionMode === 'standalone'
                                ? 'bg-[#1c1a15] border-[#c8a96e] text-[#f3f1ec]'
                                : 'bg-[#18181c] border-[#232329] text-[#96938a]'
                            }`}
                          >
                            <div className="text-xs font-bold text-[#e6d1a3]">صنف مستقل لوحده</div>
                            <div className="text-[10px] mt-0.5">يُقدم منفردًا بالوزن المحدد</div>
                          </button>

                          <button
                            type="button"
                            onClick={() => setPortionMode('mix')}
                            className={`touch-press min-h-[52px] p-3 rounded-xl border text-right cursor-pointer ${
                              portionMode === 'mix'
                                ? 'bg-[#1c1a15] border-[#c8a96e] text-[#f3f1ec]'
                                : 'bg-[#18181c] border-[#232329] text-[#96938a]'
                            }`}
                          >
                            <div className="text-xs font-bold text-[#e6d1a3]">خلط مع أكلة أخرى</div>
                            <div className="text-[10px] mt-0.5">دمج في طبق واحد مخصص</div>
                          </button>
                        </div>

                        {portionMode === 'mix' && (
                          <div className="space-y-2.5 pt-2 border-t border-[#232329]">
                            <label className="block text-[11px] text-[#96938a]">
                              اختر الطبق أو الصنف للخلط معه في «{getMealSlotName(state, safeSlot)}»
                              <select
                                value={mixTargetUid}
                                onChange={(e) => setMixTargetUid(e.target.value)}
                                className="mt-1.5 w-full min-h-[40px] rounded-xl border border-[#232329] bg-[#18181c] px-3 py-2 text-xs text-[#f3f1ec]"
                              >
                                <option value="new">بدء طبق مخصص جديد بهذه الإضافة</option>
                                {mixableSlotItems.map((item) => {
                                  const m = cartItemMacros(item, state.foods, state.dishes);
                                  return (
                                    <option key={item.uid} value={item.uid}>
                                      الخلط مع: {formatItemTitle(item)} ({round0(m.grams)}غ)
                                    </option>
                                  );
                                })}
                              </select>
                            </label>

                            <label className="block text-[11px] text-[#96938a]">
                              اسم الطبق المخلوط (اختياري)
                              <input
                                type="text"
                                value={customPlateName}
                                onChange={(e) => setCustomPlateName(e.target.value)}
                                placeholder="مثال: دجاج مشوي مع أرز بسمتي"
                                className="mt-1.5 w-full min-h-[40px] rounded-xl border border-[#232329] bg-[#18181c] px-3 py-2 text-xs text-[#f3f1ec] placeholder:text-[#66645e]"
                              />
                            </label>
                          </div>
                        )}
                      </div>

                    </div>

                    {/* Sticky Bottom Action */}
                    <div className="p-4 pb-[max(1rem,env(safe-area-inset-bottom))] border-t border-[#e3dfd4] bg-[#fffefb] shrink-0">
                      <button
                        type="button"
                        disabled={grams <= 0}
                        onClick={(e) => commitFoodToMeal(selectedFood, e.currentTarget)}
                        className="touch-press w-full min-h-[48px] py-3 rounded-xl font-bold text-sm text-[#141108] bg-gradient-to-l from-[#e6d1a3] to-[#c8a96e] disabled:opacity-40 flex items-center justify-center gap-2 cursor-pointer"
                      >
                        <Check className="w-4 h-4" />
                        <span>
                          {portionMode === 'mix'
                            ? `خلط وإرسال للسلة (${round1(mac.price)} دج)`
                            : `إضافة إلى السلة (${round1(mac.price)} دج)`}
                        </span>
                      </button>
                    </div>
                  </>
                );
              })()}
            </motion.aside>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
};
